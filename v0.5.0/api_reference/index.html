def __init__(
    self,
    *,
    app_id: str,
    host_namespace: str | None = None,
    project_resources=None,
    app_settings=None,
):
    self.app_id = str(app_id).strip()
    if not self.app_id:
        raise ValueError("Workspace requires a non-empty app_id.")
    self.host_namespace = str(host_namespace or "").strip() or self.app_id
    self.settings_manager = SettingsManager()
    legacy_specs = coerce_app_setting_specs(app_settings)
    if legacy_specs:
        self.settings_manager.register_app_settings(self.app_id, legacy_specs)
    logging_cfg = self.settings_manager.settings.logging
    self.logging_manager = LoggingManager(
        global_log_dir=self.settings_manager.settings.projects_db.parent / "logs",
        max_bytes=logging_cfg.max_file_size_mb * 1024 * 1024,
        backup_count=logging_cfg.backup_count,
        retention_days=logging_cfg.retention_days,
        queue_size=logging_cfg.queue_size,
        app_level=logging_cfg.app_level,
        executor_level=logging_cfg.executor_level,
        project_level=logging_cfg.project_level,
        console_level=logging_cfg.console_level,
        root_namespace=self.host_namespace,
    )
    self.logging_manager.start()
    self.app_logger = self.logging_manager.get_app_logger("core")

    self._master_db_path = self.settings_manager.settings.projects_db

    self.master_db = MasterDB(self._master_db_path)
    self.project_manager = ProjectManager(self.master_db, app_id_filter=self.app_id)
    if self.settings_manager.settings.general.prune_missing_projects:
        pruned = self.project_manager.repository.delete_missing_projects()
        if pruned:
            self.app_logger.info("Pruned %d registry entries with missing project folders.", pruned)
    self._project_resource_contract = ProjectResourceContract(
        app_id=self.app_id,
        specs=coerce_project_resource_specs(project_resources),
    )

    self._runtime_state = ProjectRuntimeState()
    self._engine = self._flow = None  # see `Workspace.engine` and `Workspace.submit`
    # see `Workspace.activate`: who put the pool of this machine aside, who is starting, one at a time
    self._aside, self._starting, self._activation = None, set(), threading.Lock()
    self.project_logger = None
    self._advanced = WorkspaceAdvancedAccess(self)

def open_project(
    self,
    project_id: str | UUID,
    *,
    cancel_running_tasks_on_switch: bool = True,
    touch_master: bool = True,
    extra_dirs: list[str] | None = None,
) -> ProjectContext:
    normalized_id = self._normalize_id(project_id)
    self._validate_project_access(normalized_id)

    if self.active_context:
        if self.active_context.id == normalized_id:
            return self.active_context
        self.close_project(cancel_running_tasks=cancel_running_tasks_on_switch)

    context = self.project_manager.load_project_by_id(
        normalized_id,
        sm=self.settings_manager,
        extra_dirs=self._merge_project_dirs(extra_dirs),
    )
    self._activate_context(context, touch_master=touch_master)
    self.app_logger.info(
        "Project activated: id=%s name=%s",
        context.id,
        context.name,
        extra={"project_id": str(context.id)},
    )
    return context

def create_project(
    self,
    name: str,
    folder: Path | str,
    description: str = "",
    tags: list[str] | None = None,
    base_settings: SettingsProfile = "global",
    scope: str = "full",
    activate: bool = True,
    extra_dirs: list[str] | None = None,
) -> ProjectContext:
    folder = Path(folder).expanduser().resolve()
    normalized_scope = str(scope).strip()
    effective_scope = normalized_scope or "full"
    if effective_scope == "full":
        effective_scope = self.app_id
    context = self.project_manager.create_project(
        name=name,
        folder=folder,
        sm=self.settings_manager,
        base=base_settings,
        description=description,
        scope=effective_scope,
        app_id=self.app_id,
        tags=tags,
        extra_dirs=self._merge_project_dirs(extra_dirs),
    )
    if not activate:
        return context

    if self.active_context and self.active_context.id != context.id:
        self.close_project(cancel_running_tasks=True)
    self._activate_context(context, touch_master=True)
    self.app_logger.info(
        "Project created and activated: id=%s name=%s",
        context.id,
        context.name,
        extra={"project_id": str(context.id)},
    )
    return context

def create_or_open_project(
    self,
    *,
    name: str,
    folder: Path | str,
    description: str = "",
    tags: list[str] | None = None,
    base_settings: SettingsProfile = "global",
    scope: str = "full",
    cancel_running_tasks_on_switch: bool = True,
    activate: bool = True,
    extra_dirs: list[str] | None = None,
) -> ProjectContext:
    folder_path = Path(folder).expanduser().resolve()
    existing_at_path = self.project_manager.find_project_global(folder=folder_path)
    if existing_at_path is not None:
        existing_app_id = (existing_at_path.app_id or "").strip()
        if existing_app_id != self.app_id:
            raise ValueError(
                f"Path '{folder_path}' already belongs to project '{existing_at_path.name}' "
                f"of app_id='{existing_app_id or '<empty>'}'."
            )
        if activate:
            return self.open_project(
                existing_at_path.id,
                cancel_running_tasks_on_switch=cancel_running_tasks_on_switch,
                touch_master=True,
                extra_dirs=self._merge_project_dirs(extra_dirs),
            )
        return self.project_manager.load_project(
            project=existing_at_path,
            sm=self.settings_manager,
            extra_dirs=self._merge_project_dirs(extra_dirs),
        )
    existing = self.project_manager.find_project(name=name)
    if existing is not None:
        if activate:
            return self.open_project(
                existing.id,
                cancel_running_tasks_on_switch=cancel_running_tasks_on_switch,
                touch_master=True,
                extra_dirs=self._merge_project_dirs(extra_dirs),
            )
        return self.project_manager.load_project(
            project=existing,
            sm=self.settings_manager,
            extra_dirs=self._merge_project_dirs(extra_dirs),
        )
    return self.create_project(
        name=name,
        folder=folder_path,
        description=description,
        tags=tags,
        base_settings=base_settings,
        scope=scope,
        activate=activate,
        extra_dirs=self._merge_project_dirs(extra_dirs),
    )

def register_task_canceller(self, project_id: str | UUID, canceller: Callable[[], None]):

def activate(self, name: str, on_line=None) -> None:
    """Start the executor kept as `name`, which takes jobs under that name from then on.

    Refused with `ValueError` while another active one has one of its machines or is a Ray
    too (see `blocked`). A Ray that takes this machine puts the pool of it aside until it
    is deactivated, so the pool must have no job that has not ended. A host that is not
    reached or not prepared raises, and leaves everything as it was. `on_line` hears what
    is done on a host that is being prepared. It waits for its hosts: not from a thread
    that cannot.
    """
    settings, engine = self.settings_manager.settings, self.engine
    if name not in settings.executors:
        raise ValueError(f"no executor is kept as `{name}`")
    self._starting.add(name)
    try:
        with self._activation:
            if name in engine._executors:
                return
            why = blocked(name, settings, self._active())
            if why:
                raise ValueError(f"`{name}` is not started: {why}")
            config = settings.executors[name]
            aside = config.kind == "ray" and "local" in config.hosts and "local" in engine._executors
            if aside:
                try:
                    engine.unregister("local")
                except ValueError:
                    raise ValueError(f"`{name}` takes this machine, and its pool has jobs that have not ended") from None
            try:
                host_env = self.host_env and (lambda address, busy: self.host_env(address, busy, on_line))
                executor = build_executor(name, settings, host_env)
                if executor is None:
                    raise RuntimeError(_LOGIN_NODE)
                engine.register(name, executor)
            except BaseException:
                if aside:
                    self._local_back()
                raise
            if aside:
                self._aside = name
    finally:
        self._starting.discard(name)

def db_pages(
    source: Any,
    spec: QuerySpec,
    *,
    key: str = "id",
    page_size: int = 500,
) -> Iterator[dict[str, Any]]:
    """The same rows as `db_stream`, but in keyset pages.

    Each page is its own short read (`WHERE key > last ORDER BY key LIMIT n`) instead of a single
    cursor held open for the whole feed. Feeding a million-ligand screen takes hours, and a read
    snapshot held open that long prevents SQLite from checkpointing the WAL: `project.db-wal`
    grows without bound while the job itself writes its results.

    Without a usable key (raw SQL, `offset`, or an order other than `key`) no keyset is possible:
    it falls back to `db_stream`, which returns exactly the same rows.
    """
    key_name = str(key or "").strip()
    pageable = (
        bool(spec.table)
        and bool(key_name)
        and not spec.offset
        and (not spec.fields or key_name in spec.fields)
        and (not spec.order or tuple(spec.order) == (key_name,))
    )
    if not pageable:
        yield from db_stream(source, spec, batch_size=page_size)
        return

    backend = resolve_project_store(source)
    size = max(1, int(page_size))
    remaining = None if spec.limit is None else int(spec.limit)
    cursor: Any = None
    while remaining is None or remaining > 0:
        take = size if remaining is None else min(size, remaining)
        filters = dict(spec.filters)
        if cursor is not None:
            filters[f"{key_name}__gt"] = cursor
        page = replace(spec, filters=filters, order=(key_name,), limit=take, offset=0)
        rows = backend.read_rows(_as_project_read_query(page))
        if not rows:
            return
        yield from rows
        if len(rows) < take:
            return
        cursor = rows[-1].get(key_name)
        if cursor is None:
            return  # no cursor, no progress; stopping beats repeating the page forever
        if remaining is not None:
            remaining -= len(rows)

def unregister(self, name: str) -> None:
    """Shut down the executor registered as `name`, if there is one.

    Raises `ValueError` while a job submitted to it has not ended: an executor changes
    between jobs, never under one.

    if slot is None:

submit ¶

submit(executor: str, fn, chunks, *, cpu: float = 1, gpu: float = 0, mem_mb: float = 0, window: int | None = None, max_cpu: float | None = None, priority: int = 0, total: int | None = None, name: str = '', journal: str | None = None, key: str | None = None, after=(), retries: int = 0, retry_wait: float = 1.0, timeout: float | None = None, batch: int | None = None, batched: bool = False, progress: bool | None = None, origin=None, label=None, thread: bool = False, on_result=None, on_results=None, on_error=None, on_finish=None, on_terminal=None) -> int

Run fn(payload) for every payload in chunks on executor. Returns the job id.

cpu, gpu and mem_mb are what one chunk needs; a dict payload may say otherwise for itself with _cpu_required, _gpu_required and _mem_mb, which fn never sees. max_cpu caps the job as a whole; a chunk above the cap runs alone in its job. chunks is read on its own thread. window is how many chunks may be between dispatch and delivered result, and how many more may be read ahead. Chunks that take well under a task's worth of time travel several to a task where the executor allows it; each still succeeds, fails and is recorded on its own. Left to itself the window grows with them, since such chunks are small by construction. batch settles it instead: that many chunks to a task, fewer only in the last one and where chunks with different needs meet. With batched, fn takes the list of payloads of a task and returns one result for each, in order; if it raises, all of them failed. A chunk's key is content_key(payload), or the _key of a dict payload. on_result(key, result), on_error(key, error) and on_terminal(snapshot) run on the delivery thread, in order; a chunk counts as done once on_result returns. on_results(pairs) takes its place for whoever writes results somewhere: it gets every (key, result) of the job that is waiting, up to _DELIVER_MAX, and all of them count as done once it returns. When it raises, each pair is offered again alone and only the ones that raise again fail, so it must leave nothing behind when it raises: one transaction. on_finish() is the job's last word: called once every chunk has ended, on the delivery thread, before the job counts as completed and the jobs after it start. If it raises, the job fails with that error. A job that failed, was canceled or interrupted never gets there. chunks may know more than how to be read. With remaining(), how many chunks it has yet to give, the job has a total while it runs: counted at its first chunk and every _RECOUNT seconds after, exact once the source is read. With open(state), it is called before the first read and what it returns (JSON) is kept with the job in its journal and passed back as state when the job continues; state is None the first time. progress says whether the tasks of the job are watched, for JobSnapshot.progress to count how far the running chunks are and not only the ones that ended (see executors.report). By default every task is on an executor that watches for nothing (the process pool); on another, a task of several chunks, and a task of one when the job has at most two for each worker. thread says that fn spends its time waiting for a program it starts, not in Python: an executor that keeps a worker for each task and can spare it (the process pool) runs it on a thread of this process, so no worker holds memory for it. Such a chunk is stopped through the programs it tells of (see executors.launched). label is what a chunk goes by in tasks and in a failure, in place of a short repr of its payload: a template the payload fills ("dock-R{receptor}-L{ligand}", the keys of a dict payload) or a function (payload) that returns the text. def submit(self, executor: str, fn, chunks, *, cpu: float = 1, gpu: float = 0, mem_mb: float = 0,
           window: int | None = None, max_cpu: float | None = None, priority: int = 0,
           total: int | None = None, name: str = "", journal: str | None = None, key: str | None = None,
           after=(), retries: int = 0, retry_wait: float = 1.0, timeout: float | None = None,
           batch: int | None = None, batched: bool = False, progress: bool | None = None, origin=None,
           label=None, thread: bool = False,
           on_result=None, on_results=None, on_error=None, on_finish=None, on_terminal=None) -> int:
    """Run `fn(payload)` for every payload in `chunks` on `executor`. Returns the job id.

    `cpu`, `gpu` and `mem_mb` are what one chunk needs; a dict payload may say otherwise
    for itself with `_cpu_required`, `_gpu_required` and `_mem_mb`, which `fn` never sees.
    `max_cpu` caps the job as a whole; a chunk above the cap runs alone in its job.
    `chunks` is read on its own thread. `window` is how many chunks may be between dispatch
    and delivered result, and how many more may be read ahead.
    Chunks that take well under a task's worth of time travel several to a task where the
    executor allows it; each still succeeds, fails and is recorded on its own. Left to
    itself the window grows with them, since such chunks are small by construction.
    `batch` settles it instead: that many chunks to a task, fewer only in the last one and
    where chunks with different needs meet. With `batched`, `fn` takes the list of payloads
    of a task and returns one result for each, in order; if it raises, all of them failed.
    A chunk's key is `content_key(payload)`, or the `_key` of a dict payload.
    `on_result(key, result)`, `on_error(key, error)` and `on_terminal(snapshot)` run on
    the delivery thread, in order; a chunk counts as done once `on_result` returns.
    `on_results(pairs)` takes its place for whoever writes results somewhere: it gets every
    `(key, result)` of the job that is waiting, up to `_DELIVER_MAX`, and all of them count as
    done once it returns. When it raises, each pair is offered again alone and only the ones
    that raise again fail, so it must leave nothing behind when it raises: one transaction.
    `on_finish()` is the job's last word: called once every chunk has ended, on the delivery
    thread, before the job counts as completed and the jobs after it start. If it raises, the
    job fails with that error. A job that failed, was canceled or interrupted never gets there.
    `chunks` may know more than how to be read. With `remaining()`, how many chunks it has yet
    to give, the job has a total while it runs: counted at its first chunk and every
    `_RECOUNT` seconds after, exact once the source is read. With `open(state)`, it is called
    before the first read and what it returns (JSON) is kept with the job in its journal and
    passed back as `state` when the job continues; `state` is None the first time.
    `progress` says whether the tasks of the job are watched, for `JobSnapshot.progress` to
    count how far the running chunks are and not only the ones that ended (see
    `executors.report`). By default every task is on an executor that watches for nothing (the
    process pool); on another, a task of several chunks, and a task of one when the job has at
    most two for each worker.
    `thread` says that `fn` spends its time waiting for a program it starts, not in Python:
    an executor that keeps a worker for each task and can spare it (the process pool) runs it
    on a thread of this process, so no worker holds memory for it. Such a chunk is stopped
    through the programs it tells of (see `executors.launched`).
    `label` is what a chunk goes by in `tasks` and in a failure, in place of a short repr of
    its payload: a template the payload fills (`"dock-R{receptor}-L{ligand}"`, the keys of a
    dict payload) or a function `(payload)` that returns the text. It runs here, never in a
    worker, and only for a chunk that is shown.
    `journal` names an attached journal; without it nothing about the job reaches the disk.
    `key` is what the job is called there. Submitting an interrupted or canceled job again
    under its key continues it: the chunks it had done are recognised by their own keys and
    not run again. `discard` is how such a job is given up instead.
    `origin` is whatever whoever submits wants kept with the job there (JSON): `recorded`
    gives it back in `spec`.
    `after` is the ids of the jobs that must end first; `chunks` is not read until they do,
    and the job fails if one of them fails, is canceled or is interrupted.
    A chunk that fails runs again up to `retries` times, `retry_wait` seconds after the first
    failure and twice as long after each one that follows. Failing is raising, running past
    `timeout` seconds or killing its worker; falling with somebody else's worker is not, and
    costs the chunk nothing. `on_error` hears of a chunk only once it has failed for good.
    `timeout` needs an executor that can stop a running chunk; the process pool stops all
    of its chunks to stop one, and the others start again.
    """

        raise RuntimeError("engine is closed")
    slot = self._executors.get(executor)
    if slot is None:
        raise ValueError(f"unknown executor {executor!r}; registered: {sorted(self._executors)}")
    if on_result is not None and on_results is not None:
        raise ValueError("on_result or on_results, not both")
    batch = batch or getattr(slot.impl, "batch", None)
    if (batch or batched) and not slot.packs:
        raise ValueError(f"executor {executor!r} takes one chunk to a task")
    if timeout is not None and not slot.kills:
        raise ValueError(f"executor {executor!r} cannot stop a running chunk, so its chunks cannot have a timeout")
    job = _Job()
    job.journal = job.n = job.uid = job.spec = job.skip = job.source = job.key = None
    job.retries, job.retry_wait, job.timeout, job.retried = retries, retry_wait, timeout, 0
    job.after = [self._jobs.get(job_id) or self._ended[job_id] for job_id in after]
    if journal is not None:
        job.journal = self._journals.get(journal)
        if job.journal is None or job.journal.failed:
            raise ValueError(f"no usable journal {journal!r}; attached: {sorted(self._journals)}")
        job.uid, job.key = key or uuid.uuid4().hex, key  # `key`: what another run can ask for it by
        job.spec = json.dumps({

            "cpu": cpu, "gpu": gpu, "mem_mb": mem_mb, "window": window, "max_cpu": max_cpu, "priority": priority,
            "retries": retries, "retry_wait": retry_wait, "timeout": timeout, "batch": batch, "origin": origin})
    job.cost = (cpu, gpu, mem_mb) if slot.impl.uses_tokens else None
    job.needs = (cpu, gpu, mem_mb)
    if job.cost is not None and not self.tokens.ever_fits(job.cost):
        raise ValueError(f"a chunk needing (cpu, gpu, mem_mb)={job.cost} can never run: "
                         f"the engine has {self.tokens.total}")
    job.id, job.name, job.slot, job.fn, job.max_cpu = next(self._ids), name, slot, fn, max_cpu
    job.batch, job.batched, job.fine, job.label = batch or 0, batched, progress, label
    job.thread = thread and slot.threads
    job.window = max(window or 2 * slot.impl.capacity * (batch or 1), batch or 1)
    job.pace, job.pack, job.elastic = 0.0, batch or 1, window is None
    job.priority, job.on_result, job.on_error, job.on_terminal = priority, on_result, on_error, on_terminal
    job.on_results, job.on_finish, job.finishing = on_results, on_finish, False
    job.status, job.chunks = WAITING if job.after else RUNNING, chunks
    job.total = len(chunks) if total is None and hasattr(chunks, "__len__") else total
    job.done = job.failed = job.open = job.seq = job.peak_mem_mb = 0

            job.n, job.skip, job.source = resumed  # `skip`: how many chunks it had done
            job.done = job.skip

    job.ready, job.room, job.hungry, job.fed_all = deque(), threading.Event(), False, False
    job.started_at, job.finished_at, job.finished = time.time(), None, threading.Event()
    self._jobs[job.id] = job
    self._events.put((self._on_submit, job))
    return job.id

def tasks(self, job_id: int) -> list[TaskSnapshot]:
    """The tasks of the job that are with its executor now, the oldest first: what a monitor
    unfolds under a running job. At most a window of them, and none once the job ended."""
    job = self._jobs.get(job_id)

                                  None if said is None else min(1.0, said / n)))
    return tasks
